cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(PixelForge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
option(PIXELFORGE_EDITOR "Build native editor" ON)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
FetchContent_MakeAvailable(json)
add_library(level src/level.cpp src/serialization.cpp src/document.cpp)
target_include_directories(level PUBLIC src)
target_link_libraries(level PUBLIC nlohmann_json::nlohmann_json)
add_library(gameplay src/gameplay.cpp)
target_include_directories(gameplay PUBLIC src)
target_link_libraries(gameplay PUBLIC level)
if(MSVC)
 target_compile_options(level PRIVATE /W4 /permissive-)
 target_compile_options(gameplay PRIVATE /W4 /permissive-)
else()
 target_compile_options(level PRIVATE -Wall -Wextra -Wpedantic)
 target_compile_options(gameplay PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(level_tests tests/level_tests.cpp)
 target_link_libraries(level_tests PRIVATE level)
 add_executable(document_tests tests/document_tests.cpp)
 target_link_libraries(document_tests PRIVATE level)
 add_executable(gameplay_tests tests/gameplay_tests.cpp)
 target_link_libraries(gameplay_tests PRIVATE gameplay)
 if(MSVC)
  target_compile_options(level_tests PRIVATE /W4 /permissive-)
  target_compile_options(document_tests PRIVATE /W4 /permissive-)
  target_compile_options(gameplay_tests PRIVATE /W4 /permissive-)
 else()
  target_compile_options(level_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(document_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(gameplay_tests PRIVATE -Wall -Wextra -Wpedantic)
 endif()
 add_test(NAME level_tests COMMAND level_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples)
 add_test(NAME document_tests COMMAND document_tests)
 add_test(NAME gameplay_tests COMMAND gameplay_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples)
endif()
if(PIXELFORGE_EDITOR)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG 877399b2b2cf21e67554ed9046410f268ce1d1b2)
 FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG 4806a1924ff6181180bf5e4b8b79ab4394118875)
 FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG 2c980bb59875b0d32144a71867fbdebb2f77cd20)
 FetchContent_MakeAvailable(SDL3 imgui stb)
 add_library(editor_imgui ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer3.cpp)
 target_include_directories(editor_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
 target_link_libraries(editor_imgui PUBLIC SDL3::SDL3)
 add_executable(PixelForge src/main.cpp src/application.cpp src/editor.cpp src/rendering.cpp src/tileset.cpp)
 target_include_directories(PixelForge SYSTEM PRIVATE ${stb_SOURCE_DIR})
 target_compile_definitions(PixelForge PRIVATE PIXELFORGE_SOURCE_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
 if(MSVC)
  target_compile_options(PixelForge PRIVATE /W4 /permissive-)
 else()
  target_compile_options(PixelForge PRIVATE -Wall -Wextra -Wpedantic)
 endif()
 target_link_libraries(PixelForge PRIVATE level gameplay editor_imgui)
 add_test(NAME editor_smoke COMMAND PixelForge --smoke)
 set_tests_properties(editor_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 20)
 install(TARGETS PixelForge RUNTIME DESTINATION .)
 install(DIRECTORY assets/ DESTINATION assets)
 install(DIRECTORY examples/ DESTINATION examples)
 install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
 install(FILES "${SDL3_SOURCE_DIR}/LICENSE.txt"
         DESTINATION licenses RENAME SDL3-LICENSE.txt)
 install(FILES "${imgui_SOURCE_DIR}/LICENSE.txt"
         DESTINATION licenses RENAME Dear-ImGui-LICENSE.txt)
 install(FILES "${json_SOURCE_DIR}/LICENSE.MIT"
         DESTINATION licenses RENAME nlohmann-json-LICENSE.txt)
 install(FILES "${stb_SOURCE_DIR}/LICENSE"
         DESTINATION licenses RENAME stb-LICENSE.txt)
endif()
