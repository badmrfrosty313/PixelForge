cmake_minimum_required(VERSION 3.24)
project(PixelForge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
option(PIXELFORGE_EDITOR "Build native editor" ON)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
FetchContent_MakeAvailable(json)
add_library(level src/level.cpp src/serialization.cpp)
target_include_directories(level PUBLIC src)
target_link_libraries(level PUBLIC nlohmann_json::nlohmann_json)
include(CTest)
if(BUILD_TESTING)
 add_executable(level_tests tests/level_tests.cpp)
 target_link_libraries(level_tests PRIVATE level)
 add_test(NAME level_tests COMMAND level_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples)
endif()
if(PIXELFORGE_EDITOR)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG 877399b2b2cf21e67554ed9046410f268ce1d1b2)
 FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG 4806a1924ff6181180bf5e4b8b79ab4394118875)
 FetchContent_MakeAvailable(SDL3 imgui)
 add_library(editor_imgui ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer3.cpp)
 target_include_directories(editor_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
 target_link_libraries(editor_imgui PUBLIC SDL3::SDL3)
 add_executable(PixelForge src/main.cpp src/application.cpp src/editor.cpp src/rendering.cpp)
 target_link_libraries(PixelForge PRIVATE level editor_imgui)
 add_test(NAME editor_smoke COMMAND PixelForge --smoke)
 set_tests_properties(editor_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 20)
endif()
